কী বসাবেন

# ── Walled garden: expired client notice (foreign/hijacked Host) ──
# DNS hijack-এর কারণে expired client এলোমেলো Host নিয়ে আসে → এই
# default_server সেগুলো ধরে /expired/ notice page দেখায়।
# আসল roisp.duckdns.org উপরের named block-এ যায় (অপ্রভাবিত)।
server {
    listen 80 default_server;
    listen [::]:80 default_server;
    server_name _;

    # certbot renewal যাতে না ভাঙে
    location /.well-known/acme-challenge/ {
        root /var/www/html;
    }

    location / {
        # যেকোনো path → সবসময় /expired/ view
        rewrite ^.*$ /expired/ break;
        proxy_pass http://isp_django;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
        proxy_set_header Host roisp.duckdns.org;   # ALLOWED_HOSTS-এ আছে
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto http;
    }
}
মূল কৌশল: আপনার port-80 block-এ server_name roisp.duckdns.org (named) — তাই আসল domain ওটাতেই যায়। নতুন block-এ default_server + server_name _ — বাকি সব Host (hijacked expired traffic + OS captive probe) এটাতে পড়ে। rewrite ... break যেকোনো path-কে /expired/-এ পাঠায়।

কীভাবে apply করবেন

sudo nano /etc/nginx/sites-available/isp_bnas      # উপরের block যোগ করুন
sudo nginx -t                                       # syntax test
sudo systemctl reload nginx                         # কোনো downtime ছাড়াই reload

# আসল domain আগের মতোই HTTPS-এ redirect হওয়া উচিত

প্রথমটায় মেয়াদ শেষ আসবে, দ্বিতীয়টায় Location: https://... থাকবে — তাহলেই ঠিক আছে।
