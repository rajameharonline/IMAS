# ════════════════════════════════════════════════════════════════════
#  EXPIRED CLIENT WALLED GARDEN — সম্পূর্ণ MikroTik কনফিগারেশন
#  (Notice page redirect + App download + bKash payment + auto-refresh)
#
#  প্রতিটি MikroTik router-এ একবার ধাপে ধাপে চালান। RouterOS 6.49+.
#  WinBox → New Terminal (অথবা SSH)-এ section ধরে ধরে paste করুন।
#
#  কীভাবে কাজ করে (সংক্ষেপে):
#   • client মেয়াদ শেষ → RADIUS তাকে Blocked_Group-এ পাঠায় →
#     router তাকে "Expired PPPoE" profile-এ ফেলে + IP-টা "expired"
#     address-list-এ যোগ করে (RADIUS attribute দিয়ে, স্বয়ংক্রিয়)।
#   • firewall ঐ "expired" list দেখে: HTTP → notice page, DNS + payment
#     IP allow, বাকি সব drop।
#   • payment domain গুলো (bKash/CDN) auto-refresh script দিয়ে সবসময়
#     whitelisted থাকে — IP বদলালেও পেমেন্ট কাজ করে।
#
#  ⚠️ সব rule শুধু "expired" address-list-এর client-এর জন্য —
#     active client সম্পূর্ণ অপ্রভাবিত।
# ════════════════════════════════════════════════════════════════════


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  SECTION 0 — ভেরিয়েবল (আপনার নেটওয়ার্ক অনুযায়ী বদলান)
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# portalIP = Django/notice server-এর IP (router এখান থেকে notice page নেয়)
# gwIP     = Expired PPPoE profile-এর local-address (expired client-এর gateway)
:global portalIP "172.16.3.254"
:global gwIP     "192.168.170.1"


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  SECTION 1 — Expired PPPoE profile + IP pool (না থাকলে তৈরি করুন)
#  নাম হুবহু "Expired PPPoE" হতে হবে (RADIUS Mikrotik-Group এটাই পাঠায়)।
#  MTK_PPP-তে এটি আগে থেকেই আছে — থাকলে এই section skip করুন।
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
/ip pool add name="Expired PPPoE" ranges=192.168.170.2-192.168.170.254
/ppp profile add name="Expired PPPoE" local-address=192.168.170.1 \
    remote-address="Expired PPPoE" rate-limit=512k/512k only-one=yes \
    comment="Walled garden — expired clients"


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  SECTION 2 — RADIUS-side নোট (router-এ নয়, RADIUS server-এ)
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  RADIUS-এর Blocked_Group-এ এই attribute গুলো থাকতে হবে (admin panel/
#  radius_service.ensure_blocked_group স্বয়ংক্রিয়ভাবে সেট করে):
#     Mikrotik-Group        := Expired PPPoE   (Expired profile-এ ফেলে)
#     Mikrotik-Address-List := expired         (IP-কে "expired" list-এ যোগ)
#     Session-Timeout       := 300             (৫ মিনিটে re-auth)
#  এর ফলে expired client connect হলেই তার IP নিজে নিজে "expired"
#  address-list-এ ঢোকে — তাই নিচের firewall rule গুলো কাজ করে।
#  (RADIUS না থাকলে: /ip firewall address-list add list=expired
#   address=192.168.170.0/24 — কিন্তু তখন active client একই pool-এ থাকলে
#   ধরা পড়বে, তাই RADIUS পদ্ধতিই উত্তম।)


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  SECTION 3 — payment-allow: fixed entries (portal + API)
#  bKash/CDN IP নিচের SECTION 7 refresh script স্বয়ংক্রিয়ভাবে যোগ করে।
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
/ip firewall address-list add list=payment-allow address=$portalIP comment="WG: ISP portal"
/ip firewall address-list add list=payment-allow address=103.214.202.17 comment="WG: ISP API public"


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  SECTION 4 — NAT: expired HTTP → notice page + return path
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# expired client যেকোনো HTTP(80) সাইটে গেলে → portal-এর notice page
/ip firewall nat add chain=dstnat src-address-list=expired protocol=tcp dst-port=80 \
    action=dst-nat to-addresses=$portalIP to-ports=80 comment="WG: expired HTTP redirect"
# portal-এ যাওয়া traffic-এর reply ঠিকমতো ফিরতে masquerade
/ip firewall nat add chain=srcnat src-address-list=expired dst-address=$portalIP \
    action=masquerade comment="WG: expired portal masq"


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  SECTION 5 — Firewall filter: walled garden core (ক্রম গুরুত্বপূর্ণ)
#  accept গুলো আগে, drop সবশেষে। সব শুধু "expired" client-এর জন্য।
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
/ip firewall filter add chain=forward src-address-list=expired protocol=udp dst-port=53 \
    action=accept comment="WG: expired allow DNS udp"
/ip firewall filter add chain=forward src-address-list=expired protocol=tcp dst-port=53 \
    action=accept comment="WG: expired allow DNS tcp"
/ip firewall filter add chain=forward src-address-list=expired dst-address-list=payment-allow \
    action=accept comment="WG: expired allow payment+portal"
/ip firewall filter add chain=forward src-address-list=expired dst-address=$gwIP \
    action=accept comment="WG: expired allow gateway"
/ip firewall filter add chain=forward src-address-list=expired \
    action=drop comment="WG: expired drop rest"


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  SECTION 6 — DNS-force: expired client-এর DNS router-এ আনা
#  (app পেমেন্টের CDN domain গুলো fixed IP-তে pin করতে দরকার)
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
/ip dns set allow-remote-requests=yes
/ip firewall nat add chain=dstnat src-address-list=expired protocol=udp dst-port=53 \
    action=redirect to-ports=53 comment="WG: expired DNS"
/ip firewall nat add chain=dstnat src-address-list=expired protocol=tcp dst-port=53 \
    action=redirect to-ports=53 comment="WG: expired DNS"
/ip firewall filter add chain=input src-address-list=expired protocol=udp dst-port=53 \
    action=accept place-before=0 comment="WG: expired DNS in"
/ip firewall filter add chain=input src-address-list=expired protocol=tcp dst-port=53 \
    action=accept place-before=0 comment="WG: expired DNS in"


# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  SECTION 7 — Payment whitelist auto-refresh (bKash + CDN IP)
#  domain গুলো resolve করে DNS static-এ pin করে + payment-allow-এ যোগ করে।
#  প্রতি ৩ ঘণ্টায় re-resolve — IP বদলালেও পেমেন্ট নিজে নিজে কাজ করে।
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
/system script add name=wg-payment-refresh dont-require-permissions=no policy=read,write,test source={
    :local domains {"payment.bkash.com"; "api.paystation.com.bd"; "tokenized.pay.bka.sh"; \
        "cdn.jsdelivr.net"; "cdnjs.cloudflare.com"; "code.jquery.com"; "c.tenor.com"; \
        "fonts.googleapis.com"; "fonts.gstatic.com"; "roisp.duckdns.org"}
    /ip firewall address-list remove [/ip firewall address-list find list="payment-allow" comment~"WG-auto"]
    /ip dns static remove [/ip dns static find comment~"WG-auto"]
    :foreach d in=$domains do={
        :do {
            :local ip [:resolve $d]
            /ip dns static add name=$d address=$ip comment=("WG-auto " . $d)
            /ip firewall address-list add list="payment-allow" address=$ip comment=("WG-auto " . $d)
        } on-error={ :log warning ("wg-payment: resolve fail " . $d) }
    }
    :log info "wg-payment-refresh: payment whitelist updated"
}
/system scheduler add name=wg-payment-refresh interval=3h start-time=startup \
    on-event="/system script run wg-payment-refresh" comment="WG payment whitelist auto-refresh"
# এখনই একবার চালাই (whitelist ভরে দেয়)
/system script run wg-payment-refresh

#  SECTION 8 — যাচাই (apply-এর পর চালান)
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# payment-allow তালিকা (portal + API + WG-auto bKash/CDN দেখা যাবে):

#   /ip firewall address-list print where list=expired
# DNS pin + scheduler:
#   /ip dns static print where comment~"WG-auto"

#  SECTION 9 — ROLLBACK (সম্পূর্ণ walled garden মুছে ফেলা)
